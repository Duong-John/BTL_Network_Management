
enable
configure terminal

! ===========================
! BASIC CONFIGURATION
! ===========================
hostname ASW1-96
no ip domain-lookup
no ip routing
enable secret Pass_96@CO3151
service password-encryption

! ===========================
! VLAN CONFIGURATION
! ===========================
vlan 10
 name SALES
exit
vlan 20
 name ENGINEERING
exit
vlan 30
 name SERVER_FARM
exit
vlan 99
 name MANAGEMENT
exit
vlan 999
 name NATIVE_UNUSED
exit

! ===========================
! RAPID SPANNING TREE
! ===========================
spanning-tree mode rapid-pvst

! ===========================
! UPLINK TO DSW1
! ===========================
interface Ethernet0/0
 description TRUNK_TO_DSW1-96
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,99
 no shutdown
exit

! ===========================
! UPLINK TO DSW2
! ===========================
interface Ethernet0/1
 description TRUNK_TO_DSW2-96
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,99
 no shutdown
exit

! ===========================
! PC1 SALES - VLAN 10
! ===========================
interface Ethernet2/0
 description PC1-SALES
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
exit

! ===========================
! PC3 ENGINEERING - VLAN 20
! ===========================
interface Ethernet2/1
 description PC3-ENG
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
exit

! ===========================
! MANAGEMENT SVI
! ===========================
interface Vlan99
 description MANAGEMENT_ASW1
 ip address 10.96.99.11 255.255.255.0
 no shutdown
exit

ip default-gateway 10.96.99.254

! ===========================
! SSH MANAGEMENT
! ===========================
username admin_96 privilege 15 secret Pass_96@CO3151
ip domain-name group96.local
crypto key generate rsa modulus 2048
ip ssh version 2

line vty 0 4
 login local
 transport input ssh
exit

! ===========================
! SHUTDOWN UNUSED PORTS
! ===========================
interface range Ethernet0/2 - 3
 switchport mode access
 switchport access vlan 999
 shutdown
exit

interface range Ethernet1/0 - 3
 switchport mode access
 switchport access vlan 999
 shutdown
exit

interface range Ethernet2/2 - 3
 switchport mode access
 switchport access vlan 999
 shutdown
exit

end
copy running-config startup-config



enable
configure terminal

interface range Ethernet0/0 - 1
 switchport trunk encapsulation dot1q

 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,99
 no shutdown
exit

 shutdown
exit
